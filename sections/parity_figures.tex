\begin{figure}[H]
\centering
\includegraphics[width=0.82\textwidth]{figures/parity_sweep_acceptance.pdf}
\caption{Parity experiment A: total blocked (left) and DP gap (right) at $k=10$ as the low group's mean acceptance $\mu_L$ falls ($\mu_H=0.8$, $\sigma=0.1$). Top: \emph{saved} (rumor prevented, including shielded nodes). Bottom: \emph{converted} (saved and believes the truth). Two-group SBM with 350/150 nodes, $p_{in}=0.05$, $p_{out}=0.01$, edge probability $0.08$, 5 random rumor seeds. Seeds are chosen on 100 sampled realizations and evaluated on 200 others. DP greedy maximizes $F_\lambda$ with $\lambda=2$; $\varepsilon$-DP greedy uses $\varepsilon=0.05$; Analytic sets $k_g\propto n_g/\bar a_g$ and Proportional $k_g\propto n_g$, each taking the highest-degree nodes in each group.}
\label{fig:parity-accept}
\end{figure}
